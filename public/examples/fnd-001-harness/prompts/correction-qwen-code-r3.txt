<script>
(function () {
  var cards = Array.prototype.slice.call(document.querySelectorAll('.card'));
  var filterButtons = Array.prototype.slice.call(document.querySelectorAll('#filters button'));
  var status = document.getElementById('status');
  var filterNames = { all: 'all plants', sun: 'plants that love sun', shade: 'plants that like shade' };

  function renderPlantButton(card) {
    var btn = card.querySelector('.plant-btn');
    var planted = btn.getAttribute('data-planted') === 'true';
    var name = card.getAttribute('data-name');
    btn.textContent = '';
    btn.appendChild(document.createTextNode(planted ? 'Planted' : 'Mark as planted'));
    var hint = document.createElement('span');
    hint.className = 'sr-only';
    hint.textContent = ' – ' + name + (planted ? '. Activate to undo.' : '.');
    btn.appendChild(hint);
    card.classList.toggle('is-planted', planted);
  }

  cards.forEach(function (card) {
    renderPlantButton(card);
    card.querySelector('.plant-btn').addEventListener('click', function (e) {
      var btn = e.currentTarget;
      var planted = btn.getAttribute('data-planted') !== 'true';
      btn.setAttribute('data-planted', String(planted));
      renderPlantButton(card);
      status.textContent = card.getAttribute('data-name') +
        (planted ? ' marked as planted.' : ' no longer marked as planted.');
    });
  });

  function applyFilter(filter) {
    var shown = 0;
    cards.forEach(function (card) {
      var match = filter === 'all' || card.getAttribute('data-light') === filter;
      card.hidden = !match;
      if (match) shown++;
    });
    filterButtons.forEach(function (b) {
      b.setAttribute('aria-pressed', String(b.getAttribute('data-filter') === filter));
    });
    status.textContent = 'Showing ' + shown + ' ' + (shown === 1 ? 'plant' : 'plants') +
      ' – ' + filterNames[filter] + '.';
  }

  filterButtons.forEach(function (b) {
    b.addEventListener('click', function () {
      applyFilter(b.getAttribute('data-filter'));
    });
  });

  status.textContent = 'Showing 3 plants – all plants.';
})();
</script>
