<script>
(function () {
  var cards = Array.prototype.slice.call(document.querySelectorAll('.card'));
  var filterButtons = Array.prototype.slice.call(document.querySelectorAll('.filter-btn'));
  var status = document.getElementById('status');
  var labels = { all: 'all plants', sun: 'plants that love sun', shade: 'plants that like shade' };

  function applyFilter(filter) {
    var shown = 0;
    cards.forEach(function (card) {
      var match = filter === 'all' || card.getAttribute('data-light') === filter;
      card.hidden = !match;
      if (match) shown++;
    });
    filterButtons.forEach(function (btn) {
      btn.setAttribute('aria-pressed', String(btn.getAttribute('data-filter') === filter));
    });
    status.textContent = 'Showing ' + shown + ' of ' + cards.length + ' — ' + labels[filter] + '.';
  }

  filterButtons.forEach(function (btn) {
    btn.addEventListener('click', function () {
      applyFilter(btn.getAttribute('data-filter'));
    });
  });

  cards.forEach(function (card) {
    var btn = card.querySelector('.plant-btn');
    var name = card.querySelector('h2').textContent;
    btn.addEventListener('click', function () {
      var planted = !card.classList.contains('planted');
      card.classList.toggle('planted', planted);
      btn.classList.toggle('is-planted', planted);
      btn.textContent = planted ? 'Planted' : 'Mark as planted';
      var hint = document.createElement('span');
      hint.className = 'sr-only';
      hint.textContent = ' : ' + name;
      btn.appendChild(hint);
    });
  });

  applyFilter('all');
})();
</script>
